file(GLOB HELIX_CORE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
add_library(helix-core STATIC ${HELIX_CORE_SOURCES})
target_include_directories(helix-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(helix-core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(helix-core PRIVATE /W4)
else()
  target_compile_options(helix-core PRIVATE -Wall -Wextra -Wno-unused-parameter)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm|ARM)" AND NOT CMAKE_SYSTEM_PROCESSOR MATCHES "64")
    target_compile_options(helix-core PUBLIC -mfpu=neon-vfpv4 -mfloat-abi=hard)
  endif()
endif()
if(HELIX_FORCE_SCALAR)
  target_compile_definitions(helix-core PUBLIC HELIX_FORCE_SCALAR=1)
endif()
find_package(Threads REQUIRED)
target_link_libraries(helix-core PUBLIC Threads::Threads)
