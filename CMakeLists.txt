cmake_minimum_required(VERSION 3.20)
project(HELIX VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(HELIX_BUILD_TESTS "Build tests" ON)
option(HELIX_BUILD_DESKTOP "Build the JUCE desktop plugin (fetches JUCE)" OFF)
option(HELIX_FORCE_SCALAR "Disable SIMD paths (for testing the scalar fallback)" OFF)
option(HELIX_SANITIZE "Build with ASan/UBSan" OFF)

if(HELIX_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

add_subdirectory(helix-core)
add_subdirectory(helix-render)
add_subdirectory(helix-mpc)
if(HELIX_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(HELIX_BUILD_DESKTOP)
  add_subdirectory(helix-desktop)
endif()
